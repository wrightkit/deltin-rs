variables
{
    global:
        0: g
        1: counter
        2: arr
    player:
        0: health
}

subroutines
{
    0: bump
}

rule("Subroutine bump")
{
    event
    {
        Subroutine;
        bump;
    }
    actions
    {
        Modify Global Variable(counter, Add, 1);
        If(Compare(Global Variable(counter), ==, 10));
            Set Global Variable(counter, 0);
        End;
    }
}

rule("Reconstruction surface: variables, assignments, control flow")
{
    event
    {
        Ongoing - Global;
    }
    conditions
    {
        Compare(Global Variable(g), >=, 1);
    }
    actions
    {
        Set Global Variable(g, Add(Global Variable(counter), 5));
        Modify Global Variable(g, Add, 5);
        Modify Global Variable(g, Subtract, 2);
        Modify Global Variable(g, Multiply, 3);
        Modify Global Variable(g, Divide, 4);
        Modify Global Variable(g, Modulo, 5);
        Modify Global Variable(arr, Append To Array, 7);
        Set Global Variable(g, Vector(1, 2, 3));
        Set Global Variable(g, Array(1, 2, 3));
        Set Global Variable(g, Value In Array(Array(1, 2, 3), 1));
        Set Global Variable(g, Add(Multiply(2, 3), 1));
        Set Global Variable(g, Subtract(5, 2));
        Set Global Variable(g, Divide(6, 2));
        Set Global Variable(g, If-Then-Else(Compare(Global Variable(g), >, 0), 1, 2));
        Set Global Variable(g, Max(1, 2));
        Set Global Variable(g, Min(1, 2));
        Set Global Variable(g, Count Of(Array(1, 2, 3)));
        Set Global Variable(g, Add(Compare(Global Variable(g), ==, 1), Compare(Global Variable(g), ==, 2)));
        Set Global Variable(g, And(Compare(Global Variable(g), <, 5), Compare(Global Variable(g), >, 0)));
        Set Global Variable(g, Or(Compare(Global Variable(g), ==, 0), Compare(Global Variable(g), ==, 1)));
        Set Global Variable(g, Not(Compare(Global Variable(g), ==, 0)));
        Set Global Variable(g, Round To Integer(3.5, Up));
        Call Subroutine(bump);
        If(Compare(Global Variable(g), >, 10));
            Set Global Variable(g, 0);
        Else If(Compare(Global Variable(g), >, 5));
            Set Global Variable(g, 1);
        Else;
            Set Global Variable(g, 2);
        End;
        While(Compare(Global Variable(counter), <, 3));
            Modify Global Variable(counter, Add, 1);
        End;
        For Global Variable(counter, 0, Compare(Global Variable(counter), <, 3), 1);
            Modify Global Variable(g, Add, 1);
        End;
        Wait(0.1, Ignore Condition);
        Big Message(All Players(All Teams), Custom String("<0>", Global Variable(g)));
        Small Message(All Players(All Teams), Custom String("done"));
    }
}

rule("Player variable surface")
{
    event
    {
        Ongoing - Each Player;
        All;
        All;
    }
    actions
    {
        Set Player Variable(Event Player, health, Add(Global Variable(counter), 1));
        Modify Player Variable(Event Player, health, Add, 1);
        Set Global Variable(g, Team Of(Event Player));
        Set Global Variable(g, Add((Event Player).health, 1));
        Abort;
    }
}
